# A crawler follows only the most specific group that names it: Googlebot,
# Bingbot and the AI crawlers below do NOT inherit the rules in this first
# group. Repeat every rule that should apply to them in their own group.

# Standard crawlers
User-agent: *
Allow: /
# To be removed once internal links stop carrying ?ref= (it is moving to a
# data-ref attribute), so external ?ref= links can be crawled and
# consolidated by the canonical tag.
Disallow: /*?ref=
Disallow: /*?*&ref=
Disallow: /*?redirect_url=
Disallow: /*?*&redirect_url=

# Block traditional search engines from LLM markdown mirror pages.
# These are lightweight markdown versions of our docs/blog intended for AI
# crawlers only. Blocking prevents duplicate content issues and crawl budget waste.
# NOTE: Deploy this block only after the X-Robots-Tag noindex headers (in the
# /docs-markdown/ and /blog-markdown/ route handlers) have been live for ~2 weeks.
# If Googlebot is blocked before it can re-crawl and see the noindex directive,
# affected URLs will remain stuck in GSC "Crawled - currently not indexed".
#
# /resources/access/ and /download-gate-form are NOT disallowed: both send
# noindex, and a crawler has to be allowed to fetch a page to see that.
User-agent: Googlebot
Disallow: /docs-markdown/
Disallow: /blog-markdown/
Disallow: /*?redirect_url=
Disallow: /*?*&redirect_url=

User-agent: Bingbot
Disallow: /docs-markdown/
Disallow: /blog-markdown/
Disallow: /*?redirect_url=
Disallow: /*?*&redirect_url=
