## Services and waiting

```typescript {{ title: "ci/jobs/e2e.ts" }}
const server = await $`pnpm start`.background();
await waitForPort(3000);
await waitForHttp("http://127.0.0.1:3000/health");
await server.output();
await server.kill();

await github.waitForChecks({ names: ["vercel"] });
await github.waitForWorkflow({ workflow: "npm_test.yml" });
await step.invoke("contracts", { function: contracts, data: { sha } });
```

## Tests

```typescript {{ title: "ci/jobs/test.ts" }}
ci.job({ id: "test", from: build }, fn);
ci.job({ id: "test", from: image.job("platform/node-base") }, fn);

cache: { key: files("src/**", "test/**") }
cache: { key: files("pnpm-lock.yaml"), refresh: [{ cron: "0 3 * * *" }] }
await inngest.send(invalidateEvent("base"));

await checkout({ repo: "acme/api", ref: "main" });
await $({ retries: 2 })`pnpm test`;
await $({
  timeout: "10m",
  onTimeout: async () => {
    return $`ps auxf`;
  },
})`pnpm test`;
ci.job({ id: "node", matrix: { axes: { node: ["20", "22"] } } }, fn);
```

## Changes and triggers

```typescript {{ title: "ci/pipelines/pr.ts" }}
if (await changed("docs/**")) {
  await docsSite();
}

return ci.skip("nothing changed");

on: github.checkSuite({ branch: "main" })
on: github.push({ repo: "inngest/inngestgo", branches: ["main"] })
on: github.comment({ command: "/prerelease", minPermission: "write" })
on: ci.manual({ pipelineId: "deploy", schema })

singleton: { key: "event.data.pull_request.number", mode: "cancel" }
```
