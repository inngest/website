```typescript {{ title: "test/support.test.ts" }}
import { slackMessageEvent } from "inngest/experimental/agents";
import {
  script,
  startFakes,
  withKeys,
} from "inngest/experimental/agents/testing";
import { expect, test } from "vitest";
import { createApp } from "../inngest/app";
import { serveApp } from "./serveApp";

test("answers in the Slack thread", async () => {
  await using fakes = await startFakes();

  const app = createApp({
    model: script([
      { call: { name: "lookup", input: { id: "42" } } },
      { say: "Order 42 has shipped." },
    ]),
    slackApi: `${fakes.url}/slack/api`,
  });

  await using _server = await serveApp(
    app.inngest,
    app.agents.functions(),
  );

  await app.inngest.send({
    name: slackMessageEvent,
    data: {
      channel: "C0123ABC",
      ts: "1700000000.000100",
      text: "Where is order 42?",
      user: "U1",
    },
  });

  await expect
    .poll(() => {
      return fakes.posts({ channel: "slack", method: "chat.postMessage" });
    })
    .toHaveLength(1);

  const [post] = fakes.posts({ method: "chat.postMessage" });

  expect(post?.body.text).toBe("Order 42 has shipped.");

  const thread = withKeys(
    app.support.thread("slack:C0123ABC/1700000000.000100"),
  );

  expect((await thread.state()).answer).toBe("Order 42 has shipped.");
});
```

```typescript {{ title: "inngest/app.ts" }}
import { Inngest } from "inngest";
import {
  agent,
  type AgentModel,
  createAgents,
  kv,
  memoryKvAdapter,
  slack,
} from "inngest/experimental/agents";
import { lookup } from "./tools/lookup";

export interface AppOptions {
  model: AgentModel;
  slackApi?: string;
}

export function createApp(options: AppOptions) {
  const inngest = new Inngest({ id: "support", isDev: true });

  const agents = createAgents(inngest, { store: memoryKvAdapter() });

  const support = agents.agent(
    {
      id: "support",
      on: slack.thread({ in: ["C0123ABC"] }),
      channels: ({ event }) => {
        return {
          main: slack.channel(event, {
            in: ["C0123ABC"],
            apiUrl: options.slackApi,
            token: "xoxb-test",
          }),
        };
      },
      report: "main",
    },
    async ({ inbox, control, report }) => {
      for await (const message of inbox) {
        const result = await agent.loop({
          model: options.model,
          prompt: message,
          tools: { lookup },
          ...control,
        });

        await kv.thread.set("answer", result.text);

        await report(result.text);
      }
    },
  );

  return { inngest, agents, support };
}
```

```typescript {{ title: "inngest/tools/lookup.ts" }}
import { tool } from "inngest/experimental";
import { z } from "zod";

export const lookup = tool.step({
  description: "Find an order by ID",
  input: z.object({ id: z.string() }),
  run: async ({ id }) => {
    return { id, status: "shipped" };
  },
});
```

```typescript {{ title: "test/serveApp.ts" }}
import { createServer } from "node:http";
import type { AddressInfo } from "node:net";
import type { Inngest, InngestFunction } from "inngest";
import { serve } from "inngest/node";

export async function serveApp(
  client: Inngest.Any,
  functions: InngestFunction.Any[],
) {
  const server = createServer(serve({ client, functions }));

  await new Promise<void>((resolve) => {
    server.listen(0, resolve);
  });

  const { port } = server.address() as AddressInfo;

  await fetch(`http://localhost:${port}/api/inngest`, { method: "PUT" });

  return {
    close: async () => {
      server.close();
    },
    [Symbol.asyncDispose]: async () => {
      server.close();
    },
  };
}
```

## What it shows off

- A scripted model makes the run deterministic. [Models and the loop](/docs/labs/agents/models-and-the-loop?ref=docs-labs-agents-recipes-testing)
- A fake Slack records every post. [Reliability](/docs/labs/agents/reliability?ref=docs-labs-agents-recipes-testing)
- The same agent code runs in tests and in production. [Agents and triggers](/docs/labs/agents/agents-and-triggers?ref=docs-labs-agents-recipes-testing)

## Run it

`$DEV_SERVER` is a Dev Server with `step.subscribe`; [build it once](/docs/labs/agents/quick-start?ref=docs-labs-agents-recipes-testing#see-the-run).

```bash
$DEV_SERVER dev --no-discovery   # leave running
npx vitest run                              # in another terminal
```

The test sends events to the Dev Server at `http://localhost:8288`. This test passed against a local Dev Server.
