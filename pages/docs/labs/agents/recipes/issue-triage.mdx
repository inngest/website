# Issue triage

> An agent that labels every new issue and links the ones it has seen before.

```text
Issue #214   Dev server crashes when a step returns undefined
Labels       bug
Comment      Thanks for the report. This looks like #198, which has a fix in review.
```

## The code

<AnnotatedCode>

```typescript {{ title: "inngest/agents/triage.ts" }}
import { agent } from "inngest/experimental/agents";
import { agents } from "../client";
import { issueOpened } from "../events";
import { claude } from "../models/claude";
import { applyTriage } from "../tools/applyTriage";
import { findSimilar } from "../tools/findSimilar";

export const triage = agents.agent(
  {
    id: "triage",
    on: {
      event: issueOpened,
      thread: '"issue:" + event.data.repo + "#" + string(int(event.data.number))',
      message: (event) => {
        return `${event.data.title}\n\n${event.data.body}`;
      },
    },
  },
  async ({ inbox, trigger, control }) => {
    const issue = `${trigger?.data.repo}#${trigger?.data.number}`;

    for await (const message of inbox) {
      await agent.loop({
        model: claude,
        system:
          "Triage new issues. Search for similar open issues, then label the issue and comment once. Link duplicates.",
        prompt: `Issue ${issue}\n\n${message.content ?? ""}`,
        tools: { findSimilar, applyTriage },
        ...control,
      });
    }
  },
);
```

    number: z.number(),

```typescript {{ title: "inngest/tools/findSimilar.ts" }}
import { tool } from "inngest/experimental";
import { z } from "zod";
import { github } from "../github";

export const findSimilar = tool.step({
  description: "Search open issues in the repository for similar reports",
  input: z.object({ repo: z.string(), query: z.string() }),
  run: async ({ repo, query }) => {
    const found = (await github(
      `/search/issues?q=${encodeURIComponent(`repo:${repo} is:issue is:open ${query}`)}&per_page=5`,
    )) as { items: { number: number; title: string }[] };

    return found.items.map((item) => {
      return { number: item.number, title: item.title };
    });
  },
});
```

```typescript {{ title: "inngest/tools/applyTriage.ts" }}
import { step } from "inngest";
import { tool } from "inngest/experimental";
import { z } from "zod";
import { github } from "../github";

export const applyTriage = tool.group({
  description: "Label the issue and leave a short comment",
  input: z.object({
    repo: z.string(),
    number: z.number(),
    labels: z.array(z.enum(["bug", "feature", "question", "duplicate"])),
    comment: z.string(),
  }),
  run: async ({ repo, number, labels, comment }) => {
    await step.run("label", async () => {
      await github(`/repos/${repo}/issues/${number}/labels`, {
        method: "POST",
        body: JSON.stringify({ labels }),
      });
    });

    await step.run("comment", async () => {
      await github(`/repos/${repo}/issues/${number}/comments`, {
        method: "POST",
        body: JSON.stringify({ body: comment }),
      });
    });

    return { labeled: labels };
  },
});
```

```typescript {{ title: "inngest/github.ts" }}
export async function github(path: string, init?: RequestInit) {
  const response = await fetch(`https://api.github.com${path}`, {
    ...init,
    headers: {
      accept: "application/vnd.github+json",
      authorization: `Bearer ${process.env.GITHUB_TOKEN}`,
    },
  });

  if (!response.ok) {
    throw new Error(`GitHub ${response.status}: ${await response.text()}`);
  }

  return response.json() as Promise<unknown>;
}
```

`inngest/client.ts`, `inngest/models/claude.ts` and `inngest/serve.ts` follow the [Slack assistant](/docs/labs/agents/recipes/slack-assistant?ref=docs-labs-agents-recipes-issue-triage), with `./agents/triage` imported.

## What it shows off

- Start an agent from any event, typed. [Agents and triggers](/docs/labs/agents/agents-and-triggers?ref=docs-labs-agents-recipes-issue-triage)
- One thread per issue, so a later comment joins the same run. [Inboxes](/docs/labs/agents/inboxes?ref=docs-labs-agents-recipes-issue-triage)
- A tool that runs two steps, so a retry never labels twice. [Tools](/docs/labs/agents/tools?ref=docs-labs-agents-recipes-issue-triage)
